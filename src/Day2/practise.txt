import { useState } from "react";
function App(){
  const [count,setCount]=useState(5);

  return (
    <div>
<h1>{count}</h1>
<button onClick={()=>setCount(count+1)}>Increase</button>
<button onClick={()=>{
  if(count>0){
    setCount(count-1)
  }
}}>Decrease</button>
<button onClick={()=>setCount(5)}>Reset</button>
 </div>
  )
}
export default App;
